# Cardia target firmware -- STM32F446RE (Cortex-M4F).
#
# Configure with firmware/cmake/arm-none-eabi.cmake; the host test/simulator
# builds live elsewhere and compile the same DSP and NN sources with the system
# compiler. Nothing under src/dsp or src/nn is target-specific, which is what
# makes the host/target parity check meaningful.
#
#   cmake -S firmware -B build/arm -G Ninja \
#         -DCMAKE_TOOLCHAIN_FILE=firmware/cmake/arm-none-eabi.cmake
#   cmake --build build/arm

cmake_minimum_required(VERSION 3.20)

project(cardia
        VERSION 0.1.0
        DESCRIPTION "Real-time ECG beat classification on STM32F446RE"
        LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE MinSizeRel CACHE STRING "" FORCE)
endif()

# ---------------------------------------------------------------------------
# Options
# ---------------------------------------------------------------------------

# CMSIS-NN swaps the portable reference kernels for ARM's SIMD ones. Off by
# default so a fresh clone builds with no submodules fetched: the reference
# kernels are bit-exact with CMSIS-NN by construction (same gemmlowp
# requantisation), so this is a speed switch, never a correctness one.
option(CARDIA_USE_CMSIS_NN "Link ARM CMSIS-NN kernels instead of the portable ones" OFF)

# HIL replaces the ADC front end with an int16 sample stream over USART2, so a
# MIT-BIH record can be pushed through the physical device and diffed against
# the host simulator.
option(CARDIA_HIL_MODE "Feed samples from USART2 instead of the ADC" OFF)

set(LINKER_SCRIPT ${CMAKE_CURRENT_SOURCE_DIR}/linker/STM32F446RETx_FLASH.ld)

# ---------------------------------------------------------------------------
# Sources
# ---------------------------------------------------------------------------

# Portable: compiled identically here and in the host build.
set(CARDIA_PORTABLE_SOURCES
    src/dsp/biquad.c
    src/dsp/filter_coeffs.c
    src/dsp/pan_tompkins.c
    src/dsp/pipeline.c
    src/nn/nn_kernels.c
    src/nn/inference.c
    src/nn/cardia_model.c      # generated by ml/scripts/export_model.py
)

# Target-only: startup, clock tree, drivers, application.
set(CARDIA_TARGET_SOURCES
    src/startup_stm32f446xx.c
    src/system_stm32f4xx.c
    src/syscalls.c
    src/drivers/rcc.c
    src/drivers/gpio.c
    src/drivers/dwt.c
    src/drivers/usart.c
    src/drivers/adc_dma.c
    src/drivers/tim.c
    src/main.c
)

foreach(_src IN LISTS CARDIA_PORTABLE_SOURCES)
  if(NOT EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/${_src})
    message(FATAL_ERROR
      "Missing source: ${_src}\n"
      "If this is src/nn/cardia_model.c, the quantised weights have not been "
      "exported yet. Run:  python ml/scripts/export_model.py")
  endif()
endforeach()

add_executable(cardia.elf
    ${CARDIA_PORTABLE_SOURCES}
    ${CARDIA_TARGET_SOURCES}
)

target_include_directories(cardia.elf PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/src/dsp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/nn
    ${CMAKE_CURRENT_SOURCE_DIR}/src/drivers
    ${CMAKE_CURRENT_SOURCE_DIR}/src
    ${CMAKE_CURRENT_SOURCE_DIR}/include
)

# ---------------------------------------------------------------------------
# Flags
# ---------------------------------------------------------------------------

# -mfpu=fpv4-sp-d16 -mfloat-abi=hard: the M4F has a single-precision FPU with 16
# double-word registers. Hard-float passes floats in FPU registers rather than
# through the integer ones, which matters here because the DSP chain is entirely
# float and every filter step crosses a function boundary. The consequence is
# that the compiler emits VFP instructions freely -- including in main's
# prologue -- which is why Reset_Handler enables CP10/CP11 before calling
# anything. See the comment in startup_stm32f446xx.c.
set(CARDIA_ARCH_FLAGS
    -mcpu=cortex-m4
    -mthumb
    -mfpu=fpv4-sp-d16
    -mfloat-abi=hard
)

target_compile_options(cardia.elf PRIVATE
    ${CARDIA_ARCH_FLAGS}
    # Every function and object in its own section, so --gc-sections can drop
    # what is unreachable. On a build that links a full vector table and a set
    # of drivers this is worth several kilobytes.
    -ffunction-sections
    -fdata-sections
    -Wall
    -Wextra
    -Os
    -g3        # full debug info; it lives in the .elf and never reaches flash
)

target_compile_definitions(cardia.elf PRIVATE
    STM32F446xx
)

if(CARDIA_HIL_MODE)
  target_compile_definitions(cardia.elf PRIVATE CARDIA_HIL_MODE=1)
endif()

# ---------------------------------------------------------------------------
# CMSIS-NN (optional)
# ---------------------------------------------------------------------------

if(CARDIA_USE_CMSIS_NN)
  set(CMSIS_NN_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/CMSIS-NN)
  if(NOT EXISTS ${CMSIS_NN_DIR}/Include/arm_nnfunctions.h)
    message(FATAL_ERROR
      "CARDIA_USE_CMSIS_NN is ON but third_party/CMSIS-NN is not populated.\n"
      "Run:  git submodule update --init --depth 1 --recursive\n"
      "Or configure with -DCARDIA_USE_CMSIS_NN=OFF to use the portable kernels.")
  endif()

  target_compile_definitions(cardia.elf PRIVATE CARDIA_USE_CMSIS_NN=1)
  target_include_directories(cardia.elf PRIVATE
      ${CMSIS_NN_DIR}/Include
      ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/CMSIS-DSP/Include
      ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/CMSIS_5/CMSIS/Core/Include
  )

  file(GLOB_RECURSE CMSIS_NN_SOURCES ${CMSIS_NN_DIR}/Source/*.c)
  target_sources(cardia.elf PRIVATE ${CMSIS_NN_SOURCES})
endif()

# ---------------------------------------------------------------------------
# Link
# ---------------------------------------------------------------------------

# -specs=nano.specs  : newlib-nano, which is the small-footprint libc. Only
#                      memcpy/memset are actually reached.
# -specs=nosys.specs : supply the OS-hook stubs the libc references. Our own
#                      definitions in syscalls.c override every one that
#                      matters, so a stray malloc traps instead of failing
#                      quietly.
# --gc-sections      : discard unreached sections; pairs with -ffunction-sections.
# -Map               : the map file is the authority on the flash/RAM budget and
#                      is reviewed per build, not just when something breaks.
target_link_options(cardia.elf PRIVATE
    ${CARDIA_ARCH_FLAGS}
    -T${LINKER_SCRIPT}
    -Wl,--gc-sections
    -Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/cardia.map,--cref
    -Wl,--print-memory-usage
    -specs=nano.specs
    -specs=nosys.specs
    -nostartfiles
)

target_link_libraries(cardia.elf PRIVATE m)

# Relink when the linker script changes -- CMake does not track it otherwise,
# and a stale link after editing memory regions is a confusing way to lose an
# hour.
set_target_properties(cardia.elf PROPERTIES LINK_DEPENDS ${LINKER_SCRIPT})

# ---------------------------------------------------------------------------
# Post-build artefacts
# ---------------------------------------------------------------------------

if(NOT CMAKE_OBJCOPY)
  find_program(CMAKE_OBJCOPY arm-none-eabi-objcopy REQUIRED)
endif()
if(NOT CMAKE_SIZE)
  find_program(CMAKE_SIZE arm-none-eabi-size REQUIRED)
endif()

add_custom_command(TARGET cardia.elf POST_BUILD
    # .bin for the ST-LINK mass-storage bootloader (drag-and-drop flashing),
    # .hex for anything that wants addresses embedded in the image.
    COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:cardia.elf>
            ${CMAKE_CURRENT_BINARY_DIR}/cardia.bin
    COMMAND ${CMAKE_OBJCOPY} -O ihex   $<TARGET_FILE:cardia.elf>
            ${CMAKE_CURRENT_BINARY_DIR}/cardia.hex
    COMMAND ${CMAKE_SIZE} --format=berkeley $<TARGET_FILE:cardia.elf>
    COMMENT "Generating cardia.bin / cardia.hex and reporting size"
)
